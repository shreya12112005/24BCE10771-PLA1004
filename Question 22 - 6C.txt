import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] t = new int[n];
        for (int i = 0; i < n; i++) {
            t[i] = sc.nextInt();
        }
        sc.close();

        if (n == 1) {
            System.out.println("1 0");
            return;
        }

        int left = 0, right = n - 1;
        long sumAlice = t[left];      // total time Alice has spent
        long sumBob = t[right];       // total time Bob has spent
        int aliceCount = 1;
        int bobCount = 1;

        while (left < right) {
            if (sumAlice <= sumBob) {
                // Alice finishes (at least as early as Bob), so she takes the next bar
                left++;
                if (left == right) break;   // the only remaining bar is Bob's current one
                sumAlice += t[left];
                aliceCount++;
            } else {
                // Bob finishes earlier
                right--;
                if (right == left) break;   // the only remaining bar is Alice's current one
                sumBob += t[right];
                bobCount++;
            }
        }

        System.out.println(aliceCount + " " + bobCount);
    }
}
