import java.io.*;
import java.util.*;

public class Main {

    static class FastScanner {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st;

        String next() throws IOException {
            while (st == null || !st.hasMoreTokens())
                st = new StringTokenizer(br.readLine());
            return st.nextToken();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner();

        int xs = fs.nextInt();
        int ys = fs.nextInt();

        int n = fs.nextInt();

        int[] x = new int[n];
        int[] y = new int[n];

        for (int i = 0; i < n; i++) {
            x[i] = fs.nextInt();
            y[i] = fs.nextInt();
        }

        int[] distStart = new int[n];
        int[][] dist = new int[n][n];

        for (int i = 0; i < n; i++) {
            distStart[i] = sq(x[i] - xs) + sq(y[i] - ys);
        }

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dist[i][j] = sq(x[i] - x[j]) + sq(y[i] - y[j]);
            }
        }

        int N = 1 << n;
        int INF = Integer.MAX_VALUE / 4;

        int[] dp = new int[N];
        int[] parent = new int[N];
        int[] take1 = new int[N];
        int[] take2 = new int[N];

        Arrays.fill(dp, INF);
        dp[0] = 0;

        for (int mask = 0; mask < N; mask++) {
            if (dp[mask] == INF) continue;

            int first = -1;
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) == 0) {
                    first = i;
                    break;
                }
            }

            if (first == -1) continue;

            // Take only first
            int nmask = mask | (1 << first);
            int cost = dp[mask] + 2 * distStart[first];

            if (cost < dp[nmask]) {
                dp[nmask] = cost;
                parent[nmask] = mask;
                take1[nmask] = first;
                take2[nmask] = -1;
            }

            // Take first and another object
            for (int j = first + 1; j < n; j++) {
                if ((mask & (1 << j)) != 0) continue;

                int nm = mask | (1 << first) | (1 << j);
                int c = dp[mask] + distStart[first] + dist[first][j] + distStart[j];

                if (c < dp[nm]) {
                    dp[nm] = c;
                    parent[nm] = mask;
                    take1[nm] = first;
                    take2[nm] = j;
                }
            }
        }

        int full = N - 1;
        System.out.println(dp[full]);

        ArrayList<Integer> ans = new ArrayList<>();

        int mask = full;
        while (mask != 0) {
            ans.add(0);
            ans.add(take1[mask] + 1);
            if (take2[mask] != -1)
                ans.add(take2[mask] + 1);
            mask = parent[mask];
        }
        ans.add(0);

        Collections.reverse(ans);

        StringBuilder sb = new StringBuilder();
        for (int v : ans)
            sb.append(v).append(' ');
        System.out.println(sb.toString());
    }

    static int sq(int x) {
        return x * x;
    }
}
